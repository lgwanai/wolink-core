  <action>
    Create new file internal/services/node_service.go with the following implementation:
    
    ```go
    package services

    import (
        "os"
        "os/signal"
        "runtime"
        "syscall"
        "time"
        
        "wolink-core/internal/config"
        
        "github.com/sirupsen/logrus"
    )

    // NodeStatus represents the current status of a gateway node
    type NodeStatus struct {
        NodeID        string `json:"node_id"`
        Status        string `json:"status"`
        Uptime        string `json:"uptime"`
        Version       string `json:"version"`
        Goroutines    int    `json:"goroutines"`
        MemoryUsageMB uint64 `json:"memory_usage_mb"`
    }

    // NodeService handles node status and control operations
    type NodeService struct {
        config    *config.Config
        logger    *logrus.Logger
        startTime time.Time
        version   string
    }

    // NewNodeService creates a new NodeService
    func NewNodeService(cfg *config.Config, logger *logrus.Logger, version string) *NodeService {
        return &NodeService{
            config:    cfg,
            logger:    logger,
            startTime: time.Now(),
            version:   version,
        }
    }

    // GetStatus returns current node status
    func (s *NodeService) GetStatus() *NodeStatus {
        var m runtime.MemStats
        runtime.ReadMemStats(&m)
        
        return &NodeStatus{
            NodeID:        s.config.Node.ID,
            Status:        "healthy",
            Uptime:        time.Since(s.startTime).String(),
            Version:       s.version,
            Goroutines:    runtime.NumGoroutine(),
            MemoryUsageMB: m.Alloc / 1024 / 1024,
        }
    }

    // InitiateRestart triggers a graceful restart of the node
    // Returns immediately, actual restart happens asynchronously
    func (s *NodeService) InitiateRestart() error {
        s.logger.Info("Node restart initiated via admin API")
        
        go func() {
            // Give time for response to be sent
            time.Sleep(100 * time.Millisecond)
            
            // Send SIGTERM to self for graceful shutdown
            proc, err := os.FindProcess(os.Getpid())
            if err != nil {
                s.logger.Errorf("Failed to find process: %v", err)
                return
            }
            
            s.logger.Info("Sending SIGTERM for graceful shutdown")
            proc.Signal(syscall.SIGTERM)
        }()
        
        return nil
    }
    ```
    
    Use simple signal-based restart pattern - the actual restart should be handled by external process manager (systemd, Docker, Kubernetes).
