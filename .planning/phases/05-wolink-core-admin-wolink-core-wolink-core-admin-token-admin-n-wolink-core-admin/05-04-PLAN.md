    ```go
    package middleware

    import (
        "net/http"
        "net/http/httptest"
        "testing"
        
        "wolink-core/internal/config"
        
        "github.com/gin-gonic/gin"
        "github.com/stretchr/testify/assert"
    )

    func TestAdminTokenAuth(t *testing.T) {
        gin.SetMode(gin.TestMode)
        
        tests := []struct {
            name       string
            token      string
            header     string
            wantStatus int
        }{
            {
                name:       "valid token",
                token:      "test-admin-token-32chars-long-enough",
                header:     "test-admin-token-32chars-long-enough",
                wantStatus: http.StatusOK,
            },
            {
                name:       "invalid token",
                token:      "test-admin-token-32chars-long-enough",
                header:     "wrong-token",
                wantStatus: http.StatusUnauthorized,
            },
            {
                name:       "missing token",
                token:      "test-admin-token-32chars-long-enough",
                header:     "",
                wantStatus: http.StatusUnauthorized,
            },
            {
                name:       "admin disabled",
                token:      "",
                header:     "any-token",
                wantStatus: http.StatusForbidden,
            },
        }
        
        for _, tt := range tests {
            t.Run(tt.name, func(t *testing.T) {
                cfg := &config.Config{
                    Admin: config.AdminConfig{Token: tt.token},
                }
                
                router := gin.New()
                router.Use(AdminTokenAuth(cfg))
                router.GET("/test", func(c *gin.Context) {
                    c.Status(http.StatusOK)
                })
                
                req := httptest.NewRequest("GET", "/test", nil)
                if tt.header != "" {
                    req.Header.Set("X-Admin-Token", tt.header)
                }
                
                w := httptest.NewRecorder()
                router.ServeHTTP(w, req)
                
                assert.Equal(t, tt.wantStatus, w.Code)
            })
        }
    }
    ```
  </action>

    ```go
    package handlers

    import (
        "net/http"
        "net/http/httptest"
        "testing"
        
        "wolink-core/internal/config"
        "wolink-core/internal/services"
        
        "github.com/gin-gonic/gin"
        "github.com/sirupsen/logrus"
        "github.com/stretchr/testify/assert"
    )

    func TestNodeHandler_GetStatus(t *testing.T) {
        gin.SetMode(gin.TestMode)
        
        cfg := &config.Config{
            Node: config.NodeConfig{ID: "test-node-1"},
        }
        logger := logrus.New()
        nodeService := services.NewNodeService(cfg, logger, "1.0.0-test")
        handler := NewNodeHandler(nodeService, logger)
        
        router := gin.New()
        router.GET("/admin/node/status", handler.GetStatus)
        
        req := httptest.NewRequest("GET", "/admin/node/status", nil)
        w := httptest.NewRecorder()
        router.ServeHTTP(w, req)
        
        assert.Equal(t, http.StatusOK, w.Code)
        assert.Contains(t, w.Body.String(), "test-node-1")
        assert.Contains(t, w.Body.String(), "healthy")
    }

    func TestNodeHandler_InitiateRestart(t *testing.T) {
        gin.SetMode(gin.TestMode)
        
        cfg := &config.Config{
            Node: config.NodeConfig{ID: "test-node-1"},
        }
        logger := logrus.New()
        nodeService := services.NewNodeService(cfg, logger, "1.0.0-test")
        handler := NewNodeHandler(nodeService, logger)
        
        router := gin.New()
        router.POST("/admin/node/restart", handler.InitiateRestart)
        
        req := httptest.NewRequest("POST", "/admin/node/restart", nil)
        w := httptest.NewRecorder()
        router.ServeHTTP(w, req)
        
        assert.Equal(t, http.StatusOK, w.Code)
        assert.Contains(t, w.Body.String(), "restart initiated")
        assert.Contains(t, w.Body.String(), "graceful_shutdown")
    }
    ```
