```go
func TestGracefulShutdown_WaitsForInFlightRequests(t *testing.T) {
    requestStarted := make(chan struct{})
    requestDone := make(chan struct{})

    handler := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        close(requestStarted)
        time.Sleep(200 * time.Millisecond) // Simulate slow request
        w.WriteHeader(http.StatusOK)
        close(requestDone)
    })

    srv := &http.Server{
        Addr:    ":0",
        Handler: handler,
    }

    go srv.ListenAndServe()
    time.Sleep(10 * time.Millisecond)

    // Start a request
    go func() {
        resp, _ := http.Get("http://" + srv.Addr)
        assert.Equal(t, http.StatusOK, resp.StatusCode)
    }()

    <-requestStarted // Wait for request to start

    // Initiate shutdown
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()

    err := srv.Shutdown(ctx)
    assert.NoError(t, err)

    // Verify request completed
    select {
    case <-requestDone:
        // Success
    case <-time.After(2 * time.Second):
        t.Fatal("Request did not complete before shutdown")
    }
}
```
