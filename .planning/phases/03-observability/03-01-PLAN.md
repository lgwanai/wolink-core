    - Propagates existing X-Request-ID header if present
    - Stores request ID in gin.Context under "X-Request-ID" key
    - Adds X-Request-ID to response headers
    Cases:
      - No incoming header -> generates new UUID -> returns in response header
      - Incoming X-Request-ID -> propagates same ID -> returns in response header
  </behavior>
  <implementation>
Use gin-contrib/requestid library with custom configuration:

```go
package middleware

import (
    "github.com/gin-contrib/requestid"
    "github.com/gin-gonic/gin"
    "github.com/google/uuid"
)

// RequestID returns a middleware that generates or propagates request IDs.
// Request IDs are stored in gin.Context under "X-Request-ID" key and
// added to response headers.
func RequestID() gin.HandlerFunc {
    return requestid.New(
        requestid.WithGenerator(func() string {
            return uuid.New().String()
        }),
        requestid.WithCustomHeaderStrKey("X-Request-ID"),
    )
}
```

The library handles:
- UUID generation using the provided generator
- Header propagation from incoming requests
- Context storage and response header setting

import (

func RequestID() gin.HandlerFunc {
    return requestid.New(
        requestid.WithGenerator(func() string {
            return uuid.New().String()
        }),
        requestid.WithCustomHeaderStrKey("X-Request-ID"),
    )
}
