```go
type OCRRequest struct {
	Model          string       `json:"model"`
	File           *multipart.FileHeader `json:"file"`
	Language       string       `json:"language,omitempty"` // Optional language hint
	ResponseFormat string       `json:"response_format,omitempty"` // json or text
}

// OCRResponse OCR response
type OCRResponse struct {
	Text     string `json:"text"`
	Language string `json:"language,omitempty"`
	// For structured response (bounding boxes, confidence)
	Regions []OCRRegion `json:"regions,omitempty"`
}

// OCRRegion OCR detected region with text
type OCRRegion struct {
	Text       string    `json:"text"`
	Confidence float64   `json:"confidence,omitempty"`
	BoundingBox []int     `json:"bounding_box,omitempty"` // [x, y, width, height]
}
```
